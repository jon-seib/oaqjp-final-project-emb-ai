"""
This module runs a Flask web server for the Emotion Detection application.
"""
from flask import Flask, render_template, request
from EmotionDetection import emotion_detector

app = Flask("Emotion Detector")

@app.route("/emotionDetector")
def detect_emotion():
    """
    Analyzes the text provided in the URL arguments and returns a formatted string 
    with the emotion scores and dominant emotion.
    """
    statement = request.args.get('textToAnalyze')
    result = emotion_detector(statement)
    if result['dominant_emotion'] is None:
        return "Invalid text! Please try again!"
    formatted_string = (
        f"For the given statement, the system response is "
        f"'anger': {result['anger']}, "
        f"'disgust': {result['disgust']}, "
        f"'fear': {result['fear']}, "
        f"'joy': {result['joy']} and "
        f"'sadness': {result['sadness']}. "
        f"The dominant emotion is {result['dominant_emotion']}."
    )
    return formatted_string

@app.route("/")
def render_index_page():
    """
    Renders the index page by default.
    """
    return render_template('index.html')

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
